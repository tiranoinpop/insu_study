3. 전개 연산자와 구조 분해 (예제 11)

// src/R011_SpreadOperator.jsx (componentDidMount 안)
// 배열
var varArray1 = ['num1', 'num2', 'num3'];
var varArray2 = ['num4', 'num5', 'num6'];
var sumVarArr = [varArray1, varArray2];             // ES5 처럼 보이지만 사실 "배열 2개를 담은 배열"
console.log('sumVarArr : ' + sumVarArr);
let sumLetArr = [...varArray1, ...varArray2];       // ES6 : 요소를 풀어서 하나의 배열로
console.log('sumLetArr : ' + sumLetArr);
const [sum1, sum2, ...remain] = sumLetArr;          // 구조 분해 : 앞 2개 + 나머지
console.log('3. sum1 : ' + sum1 + ', sum2 : ' + sum2 + ', remain : ' + remain);

// 객체
var varObj1 = { key1: 'value1', key2: 'value2', key3: 'value3' };
var varObj2 = { key4: 'value4', key5: 'value5', key6: 'value6' };
var sumVarObj = Object.assign({}, varObj1, varObj2);   // ES5 : Object.assign 으로 합치기
console.log('sumVarObj : ' + JSON.stringify(sumVarObj));
let sumLetObj = { ...varObj1, ...varObj2 };            // ES6 : 전개 연산자로 합치기
console.log('sumLetObj : ' + JSON.stringify(sumLetObj));
var { key1, key2, ...remainObj } = sumLetObj;          // 객체 구조 분해 : 이름으로 + 나머지
console.log('4. key1 : ' + key1 + ', key2 : ' + key2 + ', remainObj : ' + JSON.stringify(remainObj));
// 콘솔
sumVarArr : num1,num2,num3,num4,num5,num6
sumLetArr : num1,num2,num3,num4,num5,num6
3. sum1 : num1, sum2 : num2, remain : num3,num4,num5,num6
sumVarObj : {"key1":"value1","key2":"value2","key3":"value3","key4":"value4","key5":"value5","key6":"value6"}
sumLetObj : (위와 같음)
4. key1 : value1, key2 : value2, remainObj : {"key3":"value3","key4":"value4","key5":"value5","key6":"value6"}
주의할 점 보충설명
[varArray1, varArray2] 와 [...varArray1, ...varArray2] 가 똑같이 찍힌다? '문자열' + 배열 은 배열을 쉼표로 이어 붙인 문자열로 바꿔서 둘 다 같아 보일 뿐이다.
실제로는 [[num1,num2,num3],[num4,num5,num6]] (길이 2) vs [num1, ..., num6] (길이 6). console.log(sumVarArr) 처럼 + 없이 찍거나 JSON.stringify 로 찍으면 차이가 보인다. ES5 에서 배열을 합치는 방법은 varArray1.concat(varArray2)
객체는 왜 JSON.stringify 로 찍나'문자열' + 객체 는 [object Object] 로 찍혀서 내용이 안 보이기 때문
같은 키가 겹치면뒤에 오는 것이 이긴다 (Object.assign 도 전개 연산자도 같음)
... 의 두 얼굴오른쪽(값 자리)에선 풀어헤치기(전개), 왼쪽(받는 자리)에선 나머지 모으기(rest)

👉 JavaScript 노트 10-2 (나머지 매개변수·전개 연산자), 11-8 (구조 분해), 11-9 (전개로 복사). 1단락의 getDerivedStateFromProps · setState 가 state 를 합치는 것도 { ...state, ...변경 } 와 같은 원리다.

4. class 와 prototype (예제 12)

// src/R012_Class&Prototype.jsx (componentDidMount 안)
// ES5 : 함수(생성자) + prototype 으로 "클래스처럼"
var ExamCountFunc = (function () {             // 즉시 호출 함수(IIFE)로 감싸서
    function ExamCount(num) {                  // 생성자 역할의 함수
        this.number = num;
    }
    ExamCount.prototype.showNum = function () { // 메서드는 prototype 에
        console.log('1. react_ ' + this.number);
    }
    return ExamCount;
}());
var cnt = new ExamCountFunc('200');
cnt.showNum();

// ES6 : class 문법
class ExamCountClass {
    constructor(num2) {                        // 생성자
        this.number2 = num2;
    }
    showNum() {                                // 메서드 (자동으로 prototype 에 들어감)
        console.log(`2. react_${this.number2}`);
    }
}
var cnt2 = new ExamCountClass('2hundred');
cnt2.showNum();
// 콘솔
1. react_ 200
2. react_2hundred
  • ES6 의 class 는 새로운 기능이 아니라 prototype 방식을 보기 좋게 쓴 문법(문법적 설탕, syntactic sugar) 이다. 하는 일은 같다
  • React 의 클래스형 컴포넌트 class X extends Component 도 이 ES6 class 다. extends 는 상속, super(props) 는 부모 생성자 호출
  • 보충 C++ 클래스와 비슷하게 보이지만, 메서드가 각 객체에 복사되는 게 아니라 prototype 하나를 모든 객체가 공유한다

👉 JavaScript 노트 11-3 (prototype), 10-9 (즉시 호출 함수). 보충 파일 이름에 & 가 있으면 명령줄(cmd)에서 특수문자로 해석되는 등 불편할 수 있다. R012_ClassPrototype.jsx 처럼 영문·숫자·_ 만 쓰는 게 안전하다.

5. 화살표 함수와 this (예제 13) ★

// src/R013_ArrowFunction.jsx
class R013_ArrowFunction extends Component {
    constructor(props) {
        super(props)
        this.state = {
            arrrowFunc: 'React200',      // (책 그대로 r 이 3개 : 이름이니 동작엔 문제없음)
            num: 3
        };
    }

    componentDidMount() {
        Function1(1);
        this.Function2(1, 1);
        this.Function3(1, 3);
        this.Function4();
        this.Function5(0, 2, 3);

        function Function1(num1) {                          // ① ES5 함수 선언
            return console.log(num1 + '.Es5 Function');
        }
    }

    Function2 = (num1, num2) => {                          // ② 화살표 함수 (클래스 필드)
        let num3 = num1 + num2;
        console.log(num3 + '.Arrow Function :' + this.state.arrrowFunc);
    }

    Function3() {                                            // ③ 콜백 + this 를 변수에 담기
        var this_bind = this;
        setTimeout(function () {
            console.log(this_bind.state.num + 'Es5 Callback Function nobind : ');
            console.log(this.state.arrrowFunc);              // ❌ 여기 this 는 window → 에러
        }, 100);
    }

    Function4() {                                            // ④ 콜백 + bind(this)
        setTimeout(function () {
            console.log('4. Es5 Callback Function Bind : ' + this.state.arrrowFunc);
        }.bind(this), 100)
    }

    Function5 = (num1, num2, num3) => {                    // ⑤ 콜백도 화살표 함수
        const num4 = num1 + num2 + num3;
        setTimeout(() => {
            console.log(num4 + '.Arrow Callback Function : ' + this.state.arrrowFunc);
        }, 100)
    }
    ...
}
// 콘솔 (③④⑤ 는 setTimeout 이라 100ms 뒤에 찍힌다)

2.Arrow Function :React200
3Es5 Callback Function nobind :
❌ TypeError: Cannot read properties of undefined (reading 'arrrowFunc')

콜백 안의 this 를 컴포넌트로 만드는 3가지 방법

방법어떻게결과
③ 변수에 담기바깥에서 var this_bind = this, 콜백 안에서 this_bind 사용✅ (콜백 안의 this 를 그대로 쓰면 ❌)
④ bind(this)콜백 함수에 .bind(this) 로 this 를 고정해서 넘김✅
⑤ 화살표 함수화살표 함수는 자기 this 가 없어서 바깥의 this 를 그대로 씀✅ 가장 간단, 요즘 표준
  • 일반 function 의 this 는 누가 호출했느냐로 정해진다. setTimeout 이 나중에 부르면 this 는 window → window.state 는 없으니 undefined.arrrowFunc 에서 TypeError
  • 보충 책 주석은 "undefined 가 나온다" 지만 실제로는 this.state 가 undefined 라서 그 안의 값을 읽다가 에러가 난다. setTimeout 안의 에러라 화면은 그대로 보인다
  • 클래스 안에서 Function2 = () => {} 처럼 쓰면(클래스 필드 + 화살표 함수) 메서드 안의 this 가 항상 컴포넌트다. 이벤트 핸들러에 많이 쓴다

bind 는 어디서 나왔나 : 모든 함수에 기본으로 있다

함수도 객체이고, 모든 함수는 Function.prototype 의 메서드를 물려받는다 (문자열이 toUpperCase() 를 쓰는 것과 같은 원리).

메서드하는 일
bind(this값)this 가 고정된 새 함수를 만들어 돌려준다 (바로 실행하지 않음)
call(this값, 인자1, ...)this 를 정해서 바로 실행
apply(this값, [인자들])call 과 같고 인자를 배열로
// Function4 를 풀어서 쓰면
const callback = function () { console.log(this.state.arrrowFunc) }  // ① 함수 만들기
const boundCallback = callback.bind(this)                           // ② this(컴포넌트) 고정한 새 함수
setTimeout(boundCallback, 100)                                      // ③ 그걸 넘기기

8. React 에서 jQuery 쓰기 (예제 16)

npm install jquery              # react_study 폴더에서 설치 → package.json 에 추가됨
// src/R016_Jquery.jsx
import React, { Component } from 'react'
import $ from 'jquery'                     // jquery 를 $ 라는 이름으로 가져온다

class R016_Jquery extends Component {
    input_alert = (e) => {
        var input_val = $('#inputId').val();   // id 가 inputId 인 input 의 값
        alert(input_val);
    }

    render() {
        return (
            <div>
                <h2>[THIS IS Jquery]</h2>
                <input id="inputId" name="inputName" />
                <button id="buttonId" onClick={e => this.input_alert(e)}>
                    Jquery Button
                </button>
            </div>
        )
    }
}

export default R016_Jquery;
  • 화면에서 값을 입력하면 input 태그의 value 로 저장된다. 그래서 $('#inputId').val() 로 읽을 수 있다
  • 버튼을 클릭하면 onClick 의 화살표 함수가 this.input_alert(e) 를 호출한다
  • onClick={...} : HTML 의 onclick="..." 과 달리 카멜 표기이고, 문자열이 아니라 함수를 넘긴다 보충

복습 퀴즈 (눌러서 답 확인)

1. 템플릿 리터럴의 장점 2가지는?

${ } 로 변수·식을 바로 넣을 수 있고, 엔터를 그대로 줄바꿈으로 쓸 수 있다 (\n 불필요).

2. var / let / const 의 재선언 · 재할당 가능 여부는?

var : 둘 다 가능 / let : 재선언 불가, 재할당 가능 / const : 둘 다 불가. 기본은 const, 바꿀 때만 let.

3. [arr1, arr2] 와 [...arr1, ...arr2] 의 차이는?

앞은 배열 두 개를 담은 길이 2 의 배열, 뒤는 요소를 풀어 합친 하나의 배열. '문자열' + 배열로 찍으면 둘 다 쉼표로 이어져 같아 보인다.

4. const { key1, key2, ...remainObj } = obj 의 결과는?

key1, key2 는 같은 이름의 속성 값, remainObj 는 나머지 속성들을 모은 새 객체.

5. ES6 class 와 prototype 의 관계는?

class 는 prototype 방식을 보기 좋게 쓴 문법(문법적 설탕)이다. 메서드는 prototype 에 들어가 모든 객체가 공유한다.

6. setTimeout 에 넘긴 일반 function 안에서 this.state 가 에러나는 이유는?

일반 함수의 this 는 호출한 쪽이 정하는데, setTimeout 이 부르면 this 가 window 가 된다. window.state 는 없으므로 undefined 에서 속성을 읽다가 TypeError.

7. 콜백 안에서 this 를 컴포넌트로 쓰는 3가지 방법은?

바깥에서 this 를 변수에 담기(this_bind), 콜백에 .bind(this), 화살표 함수 사용(바깥 this 를 그대로 씀).

8. bind 는 왜 모든 함수에서 쓸 수 있나? bind 와 call 의 차이는?

함수도 객체이고 모든 함수가 Function.prototype 의 메서드를 물려받기 때문. bind 는 this 가 고정된 새 함수를 돌려줄 뿐 실행하지 않고, call 은 바로 실행한다.

9. forEach 와 map 의 차이는?

forEach 는 콜백의 return 을 버리고 undefined 를 돌려준다. map 은 콜백의 return 들로 같은 길이의 새 배열을 만든다.

10. componentDidMount 에서 에러가 나면?

React 가 앱 전체를 화면에서 내려 하얀 화면이 된다. F12 Console 에서 에러를 확인한다.

11. 새 라이브러리를 설치했는데 import 에러가 계속 나면?

개발 서버를 껐다 다시 켠다 (npm run dev). 그래도 안 되면 npm run dev -- --force.
